# Phase 1: Initialization and First Commit
git init
echo "hi how r u" > new.txt
git add .
git commit -m "my new commit"

# Phase 2: Branch Creation and Fast-Forward Merge
git checkout -b branch1
echo "today is wednesday" > new2.txt
git add .
git commit -m "my another commit"
git switch main
git merge branch1

# Phase 3: Conflict Preparation
git checkout -b branch2
echo "have a great day" > abc.txt
git add .
git commit -m "changes for conflict"

git switch main
echo "have a wonderful day" > abc.txt
git add .
git commit -m "conflict"

# Phase 4: Conflict Handling and Comparison
git merge branch2        # conflict occurs
# open abc.txt and resolve conflict manually

git add abc.txt
git commit -m "resolved conflict"

# Compare branches
git diff branch2

# Optional: push to remote repository
git remote add origin <repository_url>
git push -u origin main
git push origin branch2
